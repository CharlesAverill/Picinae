Ltac pr := repeat match goal with H: ?T |- _ => idtac H ":" T; revert H end.

Definition xexit (t:trace) :=
  match t with (Addr a,_)::_ => match a with
  | 1854063 | 1854222 | 1854336 | 1854761 | 1855185 => true
  | _ => false end | _ => false end.

Definition xinvs (m d s n rsp:N) (t:trace) :=
  match t with (Addr a,st)::_ => match a with
  | 1853552 => Some (st V_MEM64 = m /\ st R_RDI = d /\ st R_RSI = s /\ st R_RDX = n /\
                     st R_RSP = rsp)
  | 1854160 | 1854272 | 1854501 | 1855091 | 1854789 | 1854667 => Some (st R_R15 = 0 -> st V_MEM64 = 0 /\ st R_RDI = 0 /\ st R_RSI = 0 /\ st R_RDX = 0 /\ st R_RCX = 0 /\ st R_R10 = 0 /\ st R_YMM0 = 0 /\ st R_YMM5 = 0 /\ st R_YMM8 = 0)
  | 1854063 | 1854222 | 1854336 | 1854761 | 1855185 => Some True
  | 1853595 | 1853903 | 1853916 | 1853936 | 1853964 => Some False
  | _ => None
  end | _ => None end.

Theorem t:
  forall st t s' x' m d s n rsp
         (ENTRY: startof t (x',s') = (Addr 1853552, st))
         (MDL: models x64typctx st)
         (MEM: st V_MEM64 = m) (D: st R_RDI = d) (S: st R_RSI = s) (Nn: st R_RDX = n)
         (RSP: st R_RSP = rsp) (BIG: 64 < n),
  satisfies_all memcpy_amd64 (xinvs m d s n rsp) xexit ((x',s')::t).
Proof.
  intros. apply prove_invs.
  simpl. rewrite ENTRY. x64_step. repeat split; assumption.
  intros.
  erewrite startof_prefix in ENTRY; try eassumption.
  eapply models_at_invariant; try eassumption. apply memcpy_welltyped. intro MDL1.
  clear - PRE MDL1 BIG. rename t1 into t.
  destruct_inv 64 PRE.
  1: admit.
  all: repeat x64_step.
  all: try (match goal with |- _ R_R15 = 0 -> _ => idtac "LOOPGOAL"; match goal with |- ?G => idtac G end; fail end).
  all: admit.
Admitted.
