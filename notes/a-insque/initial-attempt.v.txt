(* insque: AArch64 LP64 insertion, exact ordered memory effects.

   Provenance: Ubuntu glibc 2.35-0ubuntu3.15, aarch64 libc.so.6,
   SHA256 79fd4ecae5fb5fab4e17ec4f145d2b79026559dce52f01a478b0a04fef3dff06.
   ELF entry 0xe45e0, Ghidra image base 0x100000, model entry 0x1e45e0.
   Eight instructions / 32 bytes. Unmodified e08_insque_arm8 generated by
   Ghidra 11.4.2 raw SLEIGH pcode and the reviewed strict scalar replacement.
   See lifter/scalar_ports/e08_insque_arm8.manifest.json. This is an IL
   partial-correctness result, conditional on inherited lifting provenance;
   no hardware refinement, fault freedom, concurrency or termination claim.

   Initial specification/proof author: OpenAI gpt-6-astra, xhigh/default.
   Final proof producer is recorded in notes/a-insque after kernel checking.
   Template: amd64_insque_proof.v (Claude Opus 5.5); no AMD64 imports.

   C interface: void insque(void *elem, void *prev).
   X0=elem, X1=prev; a node starts with two 64-bit little-endian pointers,
   next at offset 0 and prev at offset 8. Return value is void.
   If prev=0, write two null pointers to elem. Otherwise load next from
   ORIGINAL memory at prev, write elem to prev, optionally write elem to
   next+8, then write next and prev to elem and elem+8, in that order.
   insque_exact allows ALL aliases and modular address wrap. insque_links
   additionally requires elem's fields separated from the neighbor writes;
   the two neighbor fields may coincide (both receive elem) or be disjoint.
   In particular prev=next, the singleton circular-list case, is permitted.
   insque_byte_frame preserves every byte outside the actually written fields.
   SP, X30, X19-X29, modeled vector registers and permission bits are unchanged.
   The exit is immediately before RET; insque_return executes that exact RET
   to X30, changing only PC. No stack slot or x86 return convention is assumed.

   insque_access spells out sufficient nonwrapping readable/writable ranges:
   elem[0..15], prev[0..7] if nonnull, next[8..15] if nonnull. Alignment is
   unrestricted in this IL; no tunable, CPU-feature or callee contract exists.
   Access is NOT a premise of the conditional trace theorem: modeled traces
   already include permissions. The separate typed witnesses show the usable
   empty, linear-tail and circular-neighbor contracts are inhabited; they do
   not assert whole-program execution or termination.
*)
Require Import Picinae_armv8.
Require Import e08_insque_arm8.
Import ARM8Notations.
From Stdlib Require Import NArith Lia List.
Import ListNotations.
Open Scope N.

Theorem insque_nwc : forall s2 s1, e08_insque_arm8 s1 = e08_insque_arm8 s2.
Proof. reflexivity. Qed.
Theorem insque_welltyped : welltyped_prog arm8typctx e08_insque_arm8.
Proof. exact e08_insque_arm8.welltyped. Qed.

Theorem insque_preserves_callee_saved :
  forall v, In v [R_X19; R_X20; R_X21; R_X22; R_X23; R_X24; R_X25;
                 R_X26; R_X27; R_X28; R_X29; R_SP; R_X30;
                 R_Z8; R_Z9; R_Z10; R_Z11; R_Z12; R_Z13; R_Z14; R_Z15;
                 A_READ; A_WRITE] ->
  forall_endstates e08_insque_arm8 (fun _ s _ s' => s v = s' v).
Proof.
  intros v IN. apply noassign_prog_same.
  repeat destruct IN as [<-|IN]; [prove_noassign..|destruct IN].
Qed.

Definition insque_mem (m elem prev : N) :=
  if prev =? 0 then m [Ⓠ elem := 0] [Ⓠ elem ⊕ 8 := 0]
  else let next := m Ⓠ[prev] in
       let m1 := m [Ⓠ prev := elem] in
       let m2 := if next =? 0 then m1 else m1 [Ⓠ next ⊕ 8 := elem] in
       m2 [Ⓠ elem := next] [Ⓠ elem ⊕ 8 := prev].
Definition insque_postcondition (m elem prev sp lr : N) (s : store) :=
  s V_MEM64 = insque_mem m elem prev /\ s R_SP = sp /\ s R_X30 = lr.
Definition insque_exit (t : trace) :=
  match t with (Addr 0x1e45fc, _)::_ => true | _ => false end.
Definition insque_invs (m elem prev sp lr : N) (t : trace) :=
  match t with (Addr a,s)::_ => match a with
  | 0x1e45e0 => Some (s V_MEM64 = m /\ s R_X0 = elem /\
                      s R_X1 = prev /\ s R_SP = sp /\ s R_X30 = lr)
  | 0x1e45fc => Some (insque_postcondition m elem prev sp lr s)
  | _ => None end | _ => None end.

Theorem insque_exact :
  forall s t s' x' m elem prev sp lr
    (ENTRY : startof t (x',s') = (Addr 0x1e45e0,s))
    (MDL : models arm8typctx s)
    (MEM : s V_MEM64 = m) (ELEM : s R_X0 = elem) (PREV : s R_X1 = prev)
    (SP : s R_SP = sp) (LR : s R_X30 = lr),
  satisfies_all e08_insque_arm8 (insque_invs m elem prev sp lr)
    insque_exit ((x',s')::t).
Proof.
  intros. apply prove_invs.
  simpl. rewrite ENTRY. arm8_step. repeat split; assumption.
  intros.
  erewrite startof_prefix in ENTRY; try eassumption.
  eapply models_at_invariant; try eassumption. apply insque_welltyped. intro MDL1.
  clear - PRE MDL1. rename t1 into t. destruct_inv 64 PRE.
  destruct PRE as [MEM [ELEM [PREV [SP LR]]]].
  repeat arm8_step.
  all: unfold insque_postcondition; split; [|split; assumption].
  all: unfold insque_mem; rewrite BC; try rewrite BC0;
    rewrite !setmem_mod_l;
    rewrite ?(N.add_comm 8 elem), ?(N.add_comm 8 (m Ⓠ[prev])).
  all: try (apply N.eqb_eq in BC; rewrite BC).
  all: reflexivity.
Qed.

Definition insque_ret := Move R_PC (Var R_X30) $; Jmp (Var R_PC).
Theorem insque_ret_map : forall s,
  e08_insque_arm8 s 0x1e45fc = Some (4, insque_ret).
Proof. reflexivity. Qed.
Theorem insque_return : forall s,
  exec_stmt arm8typctx s insque_ret (update arm8typctx R_PC (Some 64))
    (update s R_PC (s R_X30)) (Some (Addr (s R_X30))).
Proof.
  intro s. unfold insque_ret. eapply XSeq2.
  - apply XMove with (w:=64). apply EVar. reflexivity.
  - apply XJmp with (w:=64).
    change (eval_exp (update arm8typctx R_PC (Some 64))
      (update s R_PC (s R_X30)) (Var R_PC)
      ((update s R_PC (s R_X30)) R_PC) 64).
    apply EVar. reflexivity.
Qed.
Theorem insque_return_frame : forall s m elem prev sp lr,
  insque_postcondition m elem prev sp lr s ->
  insque_postcondition m elem prev sp lr (update s R_PC (s R_X30)) /\
  (update s R_PC (s R_X30)) R_PC = lr.
Proof. intros s m elem prev sp lr [M [S L]]. split; [repeat split|]; assumption. Qed.

Lemma insque_halves : forall elem, elem + 16 <= 2^64 ->
  ~ overlap 64 (elem ⊕ 8) 8 elem 8.
Proof.
  intros elem B [i [j [I [J E]]]].
  rewrite mp2_add_l in E.
  rewrite !N.mod_small in E by (clear - B I J; lia).
  clear - E I J. lia.
Qed.
Lemma insque_lo : forall a b n,
  ~ overlap 64 a 16 b n -> ~ overlap 64 a 8 b n.
Proof.
  intros a b n NO [i [j [I [J E]]]]. apply NO.
  exists i,j. repeat split; try assumption. lia.
Qed.
Lemma insque_hi : forall a b n,
  ~ overlap 64 a 16 b n -> ~ overlap 64 (a ⊕ 8) 8 b n.
Proof.
  intros a b n NO [i [j [I [J E]]]]. apply NO.
  exists (8+i),j. split; [lia|]. split; [exact J|].
  rewrite <- E, mp2_add_l, N.add_assoc, (N.add_comm a 8). reflexivity.
Qed.
Ltac insque_loads :=
  repeat first [rewrite getmem_setmem by discriminate
    | rewrite getmem_noverlap by (assumption || (apply noverlap_symmetry; assumption))];
  change (2^(8*8)) with (2^64) in *.

Theorem insque_empty_links : forall m elem,
  elem + 16 <= 2^64 ->
  (insque_mem m elem 0) Ⓠ[elem] = 0 /\
  (insque_mem m elem 0) Ⓠ[elem ⊕ 8] = 0.
Proof.
  intros m elem B. pose proof (insque_halves elem B) as NO.
  unfold insque_mem. split; insque_loads; reflexivity.
Qed.

(* The neighbor fields need not be disjoint when they are the same field.
   prev=next is also allowed: their two distinct embedded fields do not overlap. *)
Theorem insque_links : forall m elem prev,
  elem + 16 <= 2^64 -> prev < 2^64 -> prev <> 0 ->
  let next := m Ⓠ[prev] in
  ~ overlap 64 elem 16 prev 8 ->
  (next <> 0 -> ~ overlap 64 elem 16 (next ⊕ 8) 8 /\
    (prev = next ⊕ 8 \/ ~ overlap 64 prev 8 (next ⊕ 8) 8)) ->
  let m' := insque_mem m elem prev in
  m' Ⓠ[elem] = next /\ m' Ⓠ[elem ⊕ 8] = prev /\
  m' Ⓠ[prev] = elem /\ (next <> 0 -> m' Ⓠ[next ⊕ 8] = elem).
Proof.
  intros m elem prev EB PB PN next NO1 NO2 m'.
  assert (EL : elem < 2^64) by (clear - EB; lia).
  assert (NL : next < 2^64) by apply getmem_bound.
  pose proof (insque_halves elem EB) as E8.
  pose proof (insque_lo _ _ _ NO1) as F1.
  pose proof (insque_hi _ _ _ NO1) as F2.
  subst m'. unfold insque_mem. rewrite (proj2 (N.eqb_neq _ _) PN).
  fold next. destruct (N.eqb_spec next 0) as [NZ|NZ].
  - repeat split; try (insque_loads; apply N.mod_small; assumption).
    intros H. contradiction.
  - destruct (NO2 NZ) as [NO3 [AL|NO4]];
      pose proof (insque_lo _ _ _ NO3) as F3;
      pose proof (insque_hi _ _ _ NO3) as F4.
    + repeat split; [..|intros _];
        try (insque_loads; apply N.mod_small; assumption).
      rewrite AL. insque_loads. apply N.mod_small. exact EL.
    + repeat split; [..|intros _]; insque_loads;
        apply N.mod_small; assumption.
Qed.

(* Byte frame refers to original m and only to writes taken on this path. *)
Theorem insque_byte_frame : forall m elem prev a,
  ~ overlap 64 a 1 elem 8 ->
  ~ overlap 64 a 1 (elem ⊕ 8) 8 ->
  (prev <> 0 -> ~ overlap 64 a 1 prev 8) ->
  (prev <> 0 -> m Ⓠ[prev] <> 0 -> ~ overlap 64 a 1 ((m Ⓠ[prev]) ⊕ 8) 8) ->
  (insque_mem m elem prev) Ⓑ[a] = m Ⓑ[a].
Proof.
  intros m elem prev a E0 E8 P NXT. unfold insque_mem.
  destruct (N.eqb_spec prev 0) as [Z|NZ].
  - repeat rewrite getmem_noverlap by assumption. reflexivity.
  - specialize (P NZ). destruct (N.eqb_spec (m Ⓠ[prev]) 0) as [Z|NN].
    + repeat rewrite getmem_noverlap by assumption. reflexivity.
    + specialize (NXT NZ NN).
      repeat rewrite getmem_noverlap by assumption. reflexivity.
Qed.

Definition insque_access (s : store) (m elem prev : N) :=
  elem <> 0 /\ elem + 16 <= 2^64 /\
  (forall i, i < 16 -> mem_writable s (elem+i)) /\
  (prev <> 0 -> prev + 8 <= 2^64 /\
    (forall i, i < 8 -> mem_readable s (prev+i) /\ mem_writable s (prev+i)) /\
    (m Ⓠ[prev] <> 0 -> (m Ⓠ[prev]) + 16 <= 2^64 /\
      forall i, i < 8 -> mem_writable s ((m Ⓠ[prev])+8+i))).
Definition insque_witness_store (m prev : N) : store := fun v =>
  match v with
  | V_MEM64 => m | R_X0 => 512 | R_X1 => prev
  | R_SP => 768 | R_X30 => 4096
  | A_READ | A_WRITE => N.ones 1024
  | _ => 0 end.
Lemma insque_mask_bound : N.ones 1024 < 2^(2^64).
Proof.
  eapply N.lt_le_trans with (m:=2^1024).
  - rewrite N.ones_equiv. apply N.pred_lt. apply N.pow_nonzero. discriminate.
  - apply N.pow_le_mono_r; [discriminate|vm_compute; discriminate].
Qed.
Lemma insque_witness_typed : forall m prev,
  m < 2^(8*2^64) -> prev < 2^64 ->
  models arm8typctx (insque_witness_store m prev).
Proof.
  intros m prev MB PB v w T. destruct v;
    inversion T; subst; cbn [insque_witness_store];
    try exact MB; try exact PB; try exact insque_mask_bound;
    try (apply N.neq_0_lt_0; apply N.pow_nonzero; discriminate);
    vm_compute; reflexivity.
Qed.
Lemma insque_witness_access : forall m prev,
  prev < 1000 -> m Ⓠ[prev] < 1000 ->
  insque_access (insque_witness_store m prev) m 512 prev.
Proof.
  intros m prev P NXT. unfold insque_access.
  split; [discriminate|]. split; [vm_compute; discriminate|]. split.
  - intros i I. unfold mem_writable, ARM8Arch.mem_writable.
    cbn [insque_witness_store]. apply N.ones_spec_low. clear - I; lia.
  - intro PN. split; [clear - P; lia|]. split.
    + intros i I. unfold mem_readable, mem_writable,
        ARM8Arch.mem_readable, ARM8Arch.mem_writable.
      cbn [insque_witness_store]. split; apply N.ones_spec_low; clear - P I; lia.
    + intro NN. split; [clear - NXT; lia|]. intros i I.
      unfold mem_writable, ARM8Arch.mem_writable.
      cbn [insque_witness_store]. apply N.ones_spec_low. clear - NXT I; lia.
Qed.
Theorem insque_empty_witness : exists s,
  models arm8typctx s /\ s V_MEM64 = 0 /\ s R_X0 = 512 /\ s R_X1 = 0 /\
  insque_access s 0 512 0.
Proof.
  exists (insque_witness_store 0 0). split.
  - apply insque_witness_typed; apply N.neq_0_lt_0; apply N.pow_nonzero; discriminate.
  - split; [reflexivity|]. split; [reflexivity|]. split; [reflexivity|].
    apply insque_witness_access; vm_compute; reflexivity.
Qed.
Theorem insque_tail_witness : exists s,
  models arm8typctx s /\ s V_MEM64 = 0 /\ s R_X0 = 512 /\ s R_X1 = 256 /\
  (s V_MEM64) Ⓠ[256] = 0 /\ insque_access s 0 512 256.
Proof.
  exists (insque_witness_store 0 256). split.
  - apply insque_witness_typed.
    + apply N.neq_0_lt_0; apply N.pow_nonzero; discriminate.
    + vm_compute; reflexivity.
  - split; [reflexivity|]. split; [reflexivity|]. split; [reflexivity|].
    split; [reflexivity|]. apply insque_witness_access; vm_compute; reflexivity.
Qed.
Definition insque_circular_memory := (0 [Ⓠ 256 := 256]) [Ⓠ 264 := 256].
Theorem insque_circular_witness : exists s m,
  models arm8typctx s /\ s V_MEM64 = m /\ s R_X0 = 512 /\ s R_X1 = 256 /\
  m Ⓠ[256] = 256 /\ m Ⓠ[264] = 256 /\ insque_access s m 512 256 /\
  ~ overlap 64 512 16 256 8 /\ ~ overlap 64 512 16 264 8 /\
  ~ overlap 64 256 8 264 8.
Proof.
  exists (insque_witness_store insque_circular_memory 256), insque_circular_memory.
  assert (MB : insque_circular_memory < 2^(8*2^64)).
  { change (insque_circular_memory < memsize 64).
    unfold insque_circular_memory. apply setmem_welltyped, setmem_welltyped.
    apply N.neq_0_lt_0; apply N.pow_nonzero; discriminate. }
  split; [apply insque_witness_typed; [exact MB|vm_compute; reflexivity]|].
  split; [reflexivity|]. split; [reflexivity|]. split; [reflexivity|].
  split; [reflexivity|]. split; [reflexivity|].
  split; [apply insque_witness_access; vm_compute; reflexivity|].
  repeat split. all: intros [i [j [I [J E]]]];
    rewrite !N.mod_small in E by (clear - I J; lia); clear - I J E; lia.
Qed.
