Theorem memcpy_medium_correct:
  forall st t st' x' m d s n
    (ENTRY: startof t (x',st') = (Addr 0x19ab40,st))
    (MDL: models arm8typctx st)
    (MEM: st V_MEM64 = m) (DEST: st R_X0 = d)
    (SRC: st R_X1 = s) (COUNT: st R_X2 = n)
    (DOMAIN: memcpy_medium_domain st d s n),
  satisfies_all e08___memcpy_generic_arm8
    (memcpy_medium_invs st m d s n) memcpy_medium_exit ((x',st')::t).
Proof.
  intros. destruct DOMAIN as [MEDIUM [DN [SN [NOV [READ WRITE]]]]].
  apply prove_invs.
  - simpl. rewrite ENTRY. arm8_step.
    unfold memcpy_medium_input. repeat split; try assumption.
  - intros.
    erewrite startof_prefix in ENTRY; try eassumption.
    eapply models_at_invariant; try eassumption.
    apply memcpy_medium_welltyped. intro MDL1.
    clear - PRE MDL1 MEDIUM. rename t1 into t.
    destruct_inv 64 PRE.
    unfold memcpy_medium_input in PRE.
    destruct PRE as [MEM [DEST [SRC [COUNT FRAME]]]].
    destruct (N.leb_spec n 64) as [LO|HI].
    + eapply medium_path_33_64; [clear - MEDIUM LO; lia|exact MDL1|].
      exact (conj MEM (conj DEST (conj SRC (conj COUNT FRAME)))).
    + destruct (N.leb_spec n 96) as [LO|TOP].
      * eapply medium_path_65_96; [clear - MEDIUM HI LO; lia|exact MDL1|].
        exact (conj MEM (conj DEST (conj SRC (conj COUNT FRAME)))).
      * eapply medium_path_97_128; [clear - MEDIUM TOP; lia|exact MDL1|].
        exact (conj MEM (conj DEST (conj SRC (conj COUNT FRAME)))).

Qed.
Print Assumptions memcpy_medium_correct.

(* A trace-scoped exported frame, derived only from the actual generic entry
   theorem. A global noassign assertion would be false for this union map. *)
Theorem memcpy_medium_preserves_callee_saved:
  forall st t st' a m d s n,
    startof t (Addr a,st') = (Addr 0x19ab40,st) ->
    models arm8typctx st -> st V_MEM64 = m -> st R_X0 = d ->
    st R_X1 = s -> st R_X2 = n -> memcpy_medium_domain st d s n ->
    exec_prog e08___memcpy_generic_arm8 ((Addr a,st')::t) ->
    unterminated memcpy_medium_exit t ->
    In a [0x19abf8; 0x19ac38] ->
    memcpy_medium_frame st st'.
Proof.
  intros st t st' a m d s n ENTRY MDL MEM D S NN DOMAIN XP UT RET.
  pose proof (memcpy_medium_correct st t st' (Addr a) m d s n
    ENTRY MDL MEM D S NN DOMAIN) as SA.
  pose proof (satall_trueif_inv _ _ _ _ SA XP UT) as POST.
  repeat destruct RET as [<-|RET]; try contradiction;
    cbn [memcpy_medium_invs trueif_inv] in POST;
    exact (proj2 (proj2 POST)).
Qed.
