    eapply models_at_invariant; [exact (proj1 INPUT)|apply search_welltyped|exact ENTRY|].

          clear - SPLB SPUB K; lia. }

        search_stack_image, image20, write_field32.

        f_equal; [|reflexivity].
      all: idtac "PUSH"; match goal with |- ?G => idtac G end.
      all: fail "push diagnostic".
    + destruct PRE as [[MEM [SP FIX]] R0].
