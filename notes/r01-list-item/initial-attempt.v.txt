Open Scope N.

Definition item_memory (s0:store) :=
  write_field32 (s0 V_MEM32) (s0 R_R0) item_container_offset 0.

Definition item_register_frame (s0 s:store) :=
  forall v, v <> V_MEM32 -> v <> R_R3 -> v <> F_N -> v <> F_Z ->
    s v = s0 v.

Definition item_post (s0 s:store) :=
  s V_MEM32 = item_memory s0 /\ s R_R3 = 0 /\
  s F_N = 0 /\ s F_Z = 1 /\ item_register_frame s0 s.

Definition item_exit (t:trace) :=
  match t with (Addr a,_)::_ => a =? 0x5a | _ => false end.

Definition item_invs (s0:store) (t:trace) : option Prop :=
  match t with (Addr a,s)::_ => match a with
  | 0x56 => Some (s = s0)
  | 0x5a => Some (item_post s0 s)
  | _ => None end | _ => None end.

Definition item_writable (s:store) :=
  forall i, i < 4 -> mem_writable s ((s R_R0 + 16 + i) mod 2^32).

Definition ordinary_item_lr (lr:N) := lr < 4294967280 /\ lr mod 2 = 1.

Definition item_entry (s:store) :=
  models armv6mtypctx s /\ object32 (s R_R0) list_item_size /\
  item_writable s /\ ordinary_item_lr (s R_LR).

Theorem item_memory_fields : forall m p,
  object32 p list_item_size ->
  item_contents (write_field32 m p 16 0) p
    (field32 m p 0) (field32 m p 4) (field32 m p 8) (field32 m p 12) 0.
Proof.
  intros m p OBJ.
  assert (B: p+20 <= 2^32) by exact (proj2 (proj2 (proj2 OBJ))).
  unfold item_contents, item_value_offset, item_next_offset,
    item_previous_offset, item_owner_offset, item_container_offset.
  repeat split.
  all: first [apply field32_read_write; apply mp2_gt_0 |
    apply field32_frame; apply (fields20_separate p _ _ B);
    [simpl; tauto | simpl; tauto | discriminate]].
Qed.

Theorem item_memory_outside : forall m p,
  outside_frame32 m (write_field32 m p 16 0) (p+16) 4.
Proof.
  intros m p a NO. unfold write_field32.
  apply getmem_noverlap. exact NO.
Qed.

Theorem item_preserves_registers : forall v,
  v <> V_MEM32 -> v <> R_R3 -> v <> F_N -> v <> F_Z ->
  forall_endstates vListInitialiseItem_armv6m_prog (fun _ s _ s' => s v = s' v).

  intros s a. unfold vListInitialiseItem_armv6m_prog.

Theorem item_exact : forall s0 t s' x',
  startof t (x',s') = (Addr 0x56,s0) -> models armv6mtypctx s0 ->
  satisfies_all vListInitialiseItem_armv6m_prog (item_invs s0) item_exit ((x',s')::t).
Proof.
  intros s0 t s' x' ENTRY MDL. apply prove_invs.
  - simpl. rewrite ENTRY. armv6m_step. reflexivity.
  - intros.
    erewrite startof_prefix in ENTRY; try eassumption.
    eapply models_at_invariant; try eassumption. apply item_welltyped. intro MDL1.
    clear - PRE MDL1. destruct_inv 32 PRE.
    subst s1. repeat armv6m_step.
    unfold item_post, item_memory, write_field32, item_container_offset.
    rewrite ?setmem_mod_l. repeat split; try reflexivity.
    unfold item_register_frame. intros v MEM R3 FN FZ.
    repeat rewrite update_frame by assumption. reflexivity.
Qed.

Theorem item_correct : forall s0 t s',
  startof t (Addr 0x5a,s') = (Addr 0x56,s0) -> item_entry s0 ->
  exec_prog vListInitialiseItem_armv6m_prog ((Addr 0x5a,s')::t) ->
  unterminated item_exit t ->
  item_post s0 s' /\
  item_contents (s' V_MEM32) (s0 R_R0)
    (field32 (s0 V_MEM32) (s0 R_R0) 0)
    (field32 (s0 V_MEM32) (s0 R_R0) 4)
    (field32 (s0 V_MEM32) (s0 R_R0) 8)
    (field32 (s0 V_MEM32) (s0 R_R0) 12) 0 /\
  outside_frame32 (s0 V_MEM32) (s' V_MEM32) (s0 R_R0+16) 4.
Proof.
  intros s0 t s' ENTRY [MDL [OBJ [WR LR]]] XP UT.
  pose proof (satall_trueif_inv _ _ _ _
    (item_exact s0 t s' (Addr 0x5a) ENTRY MDL) XP UT) as POST.
  change (item_post s0 s') in POST. split; [exact POST|].
  destruct POST as [MEM _]. rewrite MEM. split.
  - apply item_memory_fields. exact OBJ.
  - apply item_memory_outside.
Qed.

(* A witness with arbitrary original memory and arbitrary initial field values.
   The all-writable mask is symbolic: never compute its 2^32 bits. *)
Definition item_entry_store (m p:N) : store :=
  update (update (update (update (memory_store32 m)
    A_WRITE (N.ones (2^32))) R_R0 p) R_SP 8192) R_LR 257.

Theorem item_entry_store_witness : forall m p,
  m < memsize 32 -> object32 p list_item_size ->
  item_entry (item_entry_store m p) /\ (item_entry_store m p) V_MEM32 = m.
Proof.
  intros m p MT OBJ. split; [unfold item_entry; split|reflexivity].
  - unfold item_entry_store.
    apply models_update; [change (257 < 2^32); reflexivity|].
    apply models_update; [change (8192 < 2^32); reflexivity|].
    apply models_update; [exact (proj1 (proj2 (proj2 OBJ)))|].
    apply models_update; [apply ones_bound|].
    apply memory_store32_welltyped. exact MT.
  - split; [exact OBJ|]. split.
    + intros i I. change (N.testbit (N.ones (2^32)) ((p+16+i) mod 2^32) = true).
      rewrite N_ones_spec_ltb. apply N.ltb_lt. apply N.mod_lt. discriminate.
    + split; reflexivity.
Qed.

Theorem item_arbitrary_fields_witness : forall p value next previous owner container,
  object32 p list_item_size -> value < 2^32 -> next < 2^32 ->
  previous < 2^32 -> owner < 2^32 -> container < 2^32 ->
  exists s, item_entry s /\ s R_R0 = p /\
    item_contents (s V_MEM32) p value next previous owner container.
Proof.
  intros p value next previous owner container OBJ BV BN BP BO BC.
  set (m := image20 0 p value next previous owner container).
  assert (MT: m < memsize 32).
  { apply image20_welltyped.
    assert (Z: forall w, 0 < memsize w).
    { intro w. change (0 < 2^(2^w*8)). apply mp2_gt_0. }
    exact (Z 32). }
  exists (item_entry_store m p). split.
  - exact (proj1 (item_entry_store_witness m p MT OBJ)).
  - split; [reflexivity|]. change (item_contents m p value next previous owner container).
    apply image20_fields; try assumption. exact (proj2 (proj2 (proj2 OBJ))).
Qed.

(* Explicit execution states refer to the actual lifted instructions. *)
Definition item_after_mov (s:store) :=
  reset_temps s (update (update (update s F_N 0) F_Z 1) R_R3 0).
Definition item_after_store (s:store) :=
  let sm := item_after_mov s in
  reset_temps sm (update sm V_MEM32
    (setmem 32 LittleE 4 (s V_MEM32) ((s R_R0+16) mod 2^32) 0)).

Theorem item_mov_step : forall s,
  can_step vListInitialiseItem_armv6m_prog
    ((Addr 0x58,item_after_mov s),(Addr 0x56,s)).
Proof.
  intro s. unfold item_after_mov.
  eapply CanStep with (sz:=2) (x:=None); [reflexivity|].
  eapply XSeq2.
  - eapply XSeq2.
    + apply XMove. change (eval_exp armv6mtypctx s
        (Cast CAST_HIGH 1 (Word 0 32)) (cast CAST_HIGH 32 1 0) 1).
      apply ECast. constructor.
    + apply XMove. change (eval_exp _ _ (BinOp OP_EQ (Word 0 32) (Word 0 32))
        (eval_binop OP_EQ 32 0 0) 1). apply EBinOp; constructor.
  - apply XMove. constructor.
Qed.

Theorem item_store_step : forall s, item_writable s ->
  can_step vListInitialiseItem_armv6m_prog
    ((Addr 0x5a,item_after_store s),(Addr 0x58,item_after_mov s)).
Proof.
  intros s WR. unfold item_after_store.
  eapply CanStep with (sz:=2) (x:=None); [reflexivity|].
  apply XMove. eapply EStore.
  - apply EVar. reflexivity.
  - apply EBinOp; [apply EVar; reflexivity|constructor].
  - apply EVar. reflexivity.
  - intros i I. change (mem_writable s ((((s R_R0+16) mod 2^32)+i) mod 2^32)).
    rewrite mp2_add_l. apply WR. exact I.
Qed.

Theorem item_execution_witness : forall s, item_entry s ->
  exec_prog vListInitialiseItem_armv6m_prog
    [(Addr 0x5a,item_after_store s); (Addr 0x58,item_after_mov s); (Addr 0x56,s)] /\
  item_post s (item_after_store s).
Proof.
  intros s ENTRY.
  assert (XP: exec_prog vListInitialiseItem_armv6m_prog
    [(Addr 0x5a,item_after_store s); (Addr 0x58,item_after_mov s); (Addr 0x56,s)]).
  { apply exec_prog_step.
    - apply exec_prog_step; [apply exec_prog_none|apply item_mov_step].
    - apply item_store_step. exact (proj1 (proj2 (proj2 ENTRY))). }
  split; [exact XP|].
  exact (proj1 (item_correct s [(Addr 0x58,item_after_mov s); (Addr 0x56,s)]
    (item_after_store s) eq_refl ENTRY XP ltac:(repeat constructor))).
Qed.

Theorem item_return_step : forall s,
  ordinary_item_lr (s R_LR) ->
  can_step vListInitialiseItem_armv6m_prog
    ((Addr (N.land (s R_LR) (N.ones 32-1)),s),(Addr 0x5a,s)).
Proof.
  intros s [BOUND ODD].
  assert (RST: reset_temps s s = s).
  { extensionality v. destruct v; reflexivity. }
  rewrite <- RST at 1.
  eapply CanStep with (sz:=2) (x:=Some (Addr (N.land (s R_LR) (N.ones 32-1))));
    [reflexivity|].
  eapply XIf with (b:=1).
  - rewrite <- ODD. change (eval_exp armv6mtypctx s
      (Cast CAST_LOW 1 (Var R_LR)) (cast CAST_LOW 32 1 (s R_LR)) 1).
    apply ECast, EVar. reflexivity.
  - apply XJmp. apply EBinOp; [apply EVar; reflexivity|constructor].
Qed.

Corollary item_execution_returns : forall s, item_entry s ->
  exec_prog vListInitialiseItem_armv6m_prog
    [(Addr (N.land (s R_LR) (N.ones 32-1)),item_after_store s);
     (Addr 0x5a,item_after_store s); (Addr 0x58,item_after_mov s); (Addr 0x56,s)].
Proof.
  intros s ENTRY. apply exec_prog_step.
  - exact (proj1 (item_execution_witness s ENTRY)).
  - change (can_step vListInitialiseItem_armv6m_prog
      ((Addr (N.land (item_after_store s R_LR) (N.ones 32-1)),item_after_store s),
       (Addr 0x5a,item_after_store s))).
    apply item_return_step. exact (proj2 (proj2 (proj2 ENTRY))).
Qed.
